# Public-hygiene rules: one "<label>|<PCRE>" per line.
#
# scripts/check-public-hygiene.sh runs every line below over the tracked files,
# and scripts/check-history-hygiene.sh, in a repository that carries it, runs the
# same lines over commit messages and pull request text. Blank lines and
# whole-line `#` comments are ignored. A line with no `|`, or with an empty label
# or pattern, stops both gates with exit 2.
#
# A label may carry several patterns. Each pattern line has a fixture of its own
# in scripts/check-public-hygiene-selftest.sh that no other line catches, and the
# self-test fails when a line has none. Deleting a line, or adding one without a
# fixture, turns the self-test red.
#
# A pattern may never match a `|`: the allowlist is pipe-separated and relies on
# matched text being unable to collide with its separator.
#
# If a pattern bites honest prose, tighten it here and add the prose to
# scripts/public-hygiene-innocent-strings.txt. The allowlist is for genuine
# content that has to stay, not for a pattern that is too loose.

# Record and task ids of the private planning tracker.
private-decision-record|(?i)(?<![A-Za-z0-9])decision[-_][0-9]+
planning-task-id|(?i)(?<![A-Za-z0-9])task[-_][0-9]+

# A bare ticket reference. The guards keep Rust generics (`<T22>`, `T22: Clone`,
# `-> T22`) and ISO timestamps (`...T10:30`) out.
bare-ticket-ref|(?<![-_A-Za-z0-9<,:&(])(?<!: )(?<!-> )T[0-9]{2,3}(?![-_A-Za-z0-9>,:(])
bare-ticket-ref|(?<![-_A-Za-z0-9>])\(T[0-9]{1,3}\)

# Milestone ids. The first needs two digits and refuses a leading `/` so URL
# segments and CSS custom properties with one-digit suffixes stay clean.
milestone-id|(?i)(?<![-_A-Za-z0-9/])m-[0-9]{2,}(?![-_A-Za-z0-9])
milestone-id|(?i)(?<![A-Za-z0-9])milestones?[ _-](?:m[-_]?)?[0-9]+(?![-_A-Za-z0-9])

# Acceptance-criterion shorthand. The trailing guard refuses a following hex
# digit so a digest is not read as one. The bare form refuses a leading `+`
# because a Unicode code point is written that way, and the first Hangul syllable
# has the same four characters as a criterion number.
acceptance-criterion|(?i)(?<![A-Za-z0-9])ac ?#[0-9]+(?![0-9a-f])
acceptance-criterion|(?i)(?<![A-Za-z0-9+])ac[0-9]+(?![0-9a-f])

private-doc-id|(?i)(?<![A-Za-z0-9])doc[-_][0-9]+(?![A-Za-z])

# Card ids need three or four digits; the word on its own is left alone.
planning-card-id|(?i)(?<![A-Za-z0-9])cards?[ _-]#?[0-9]{3,4}(?![0-9])

# Spec criterion ids. The `[-_]` before `ac` keeps hex digests out.
spec-ac-id|(?i)[a-z]{2,6}[-_]ac[-_]?[0-9]+[a-z]?(?![0-9])
spec-ac-id|(?i)(?<![A-Za-z0-9])ac[-_][0-9]+[a-z]?(?![0-9])

# A card named by its title: six or more hyphen-joined words, each starting with
# a letter. Six is the floor, and five words, one segment short of it, is the
# case the innocent-strings fixture keeps below it. scripts/check-public-hygiene.sh
# excludes vendored source from this one label, because upstream documentation
# URLs carry runs of this shape that nobody here wrote.
card-slug|(?i)(?<![A-Za-z0-9-])[a-z][a-z0-9]*(?:-[a-z][a-z0-9]*){5,}(?![A-Za-z0-9-])
card-noun-reference|(?i)(?<![A-Za-z0-9])(?:this|that|the) card(?![-_A-Za-z0-9])

# Pointers into the planning vault. TOML `[[bin]]` headers carry no hyphen and
# stay clean.
vault-wikilink|(?i)\[\[[a-z0-9]+(?:-[a-z0-9]+)+\]\]
vault-ledger-path|(?i)(?<![-_A-Za-z0-9/.])(?:decisions|cards|milestones|drafts|notes)/[a-z0-9]+(?:-[a-z0-9]+){2,}

# A numbered choice record of a retired scheme. Three or four digits, so the word
# beside a small count ("choice 2 of 3") is left alone.
planning-choice-id|(?i)(?<![A-Za-z0-9])choices?[ _-]?#?[0-9]{3,4}(?![0-9])

# A spec named by its date and a slug, in lower case or with capitals. A bare
# date, a date range and a dated file name carrying an extension are left alone:
# the slug must start with a letter, run at least two words, and not be followed
# by `.ext` or `.*`.
dated-spec-slug|(?i)(?<![-_A-Za-z0-9])20[0-9]{2}-[0-9]{2}-[0-9]{2}-[a-z][a-z0-9]*(?:-[a-z0-9]+)+(?![-_A-Za-z0-9]|\.[A-Za-z0-9*])

# A numbered finding of a review. Qualified by the review that raised it, it is
# a pointer whatever follows. Unqualified, it is a pointer only when no word or
# number follows, which leaves the participle beside a count alone:
# "finding 2 matches" is ordinary English.
review-finding-ref|(?i)(?<![A-Za-z0-9])(?:review(?:[-_ ]spec)?|delta)[-_ ]finding[ -]?#?[0-9]+
review-finding-ref|(?i)(?<![A-Za-z0-9])finding[ -]?#?[0-9]+(?:/[0-9]+)*(?![0-9/])(?!\s*[A-Za-z0-9])
